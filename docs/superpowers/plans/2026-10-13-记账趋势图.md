# 记账「近 6 个月」趋势图 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 记账页左栏多一张「近 6 个月」柱状图：六个月的支出（可切收入）并排摆着，
一眼看出哪个月花得多、哪个月是空的；数字直接标在柱子上，悬停给完整金额。

**Architecture:** 数据全从现有账目现算（不落盘、不加字段），口径走现成的
`monthTotals`；三个纯函数放 `finance-calc.js`，图是**手写内联 SVG**（和环形图、
自媒体那条曲线一个路子，零依赖、颜色走 CSS 变量）。

**Tech Stack:** 原生 HTML + CSS + ES 模块（无框架、无构建、零外链）；测试用 Node
（纯逻辑）+ 无头 Edge 的 CDP 回归。

**Spec:** `docs/superpowers/specs/2026-10-13-记账趋势图-design.md`（动手前先读一遍）

## Global Constraints

- **零外链**：界面文件里不能出现任何 `http(s)://` 链接；不引图表库。
- **不装依赖**：Node 只用来跑 `tests\*.test.mjs`（直接 `node 文件名`）。
- **口径只有一处**：月度合计走 `monthTotals`，金额显示走 `fmtMoney` / `fmtMoneyShort`
  —— 不许另写一套加总，免得出现「图上 427、概览 428」。
- **颜色只用 CSS 变量**：支出 `var(--err)`、收入 `var(--ok)`，和概览、月历同一组；
  四套皮肤 × 深浅两色都不能破相。
- **不加数据字段、不落盘**：这张图完全现算。
- **事件不挂公共 `#view`**：一律 `bindFresh(root, {...})`。
- **注释与提交信息用中文**，说清「为什么」。
- 每个任务结束：`node tests\记账计算.test.mjs` 与 `python tests\自检.py` 都要过；
  碰了页面的任务还要跑 `node tests\转账.test.mjs`（它也在记账页上，是现成的冒烟）。

```js
/* ---------------- 近 6 个月趋势（2026-10-13 加的） ----------------
   记账页一直是「一个月一个月」的：月历按月翻、概览和环形图只看当前那个月。
   这一组给趋势图备数据：最近几个月各是多少、柱高按什么比例算、合计与月均。
   口径走现成的 monthTotals —— 图上和概览上的数永远是同一个来源。 */

/** "2026-10" 往回数 → ["2026-05" … "2026-10"]（升序、含当月；跨年接得上） */
export function lastMonths(month, count = 6) {
  const m = /^(\d{4})-(\d{2})$/.exec(String(month == null ? "" : month).trim());
  if (!m) return [];
  const n = Math.max(1, Math.floor(Number(count) || 1));
  const end = Number(m[1]) * 12 + (Number(m[2]) - 1);   // 以「0 年 1 月」为 0 的月序号
  const out = [];
  for (let i = n - 1; i >= 0; i -= 1) {
    const at = end - i;
    out.push(`${Math.floor(at / 12)}-${String((at % 12) + 1).padStart(2, "0")}`);
  }
  return out;
}

/**
 * 每个月那一头的数 + 给画柱用的比例。
 * ratio = 这个月 / 最高的那个月（全是 0 时都给 0，不除零）——柱高直接用它。
 */
export function monthTrend(transactions, months, type = "expense") {
  const list = (months || []).map((month) => {
    const totals = monthTotals(transactions, month);
    return { month, cents: type === "income" ? totals.incomeCents : totals.expenseCents };
  });
  const max = list.reduce((best, row) => Math.max(best, row.cents), 0);
  return list.map((row) => ({ ...row, ratio: max ? row.cents / max : 0 }));
}

/** 卡片头上那句：几个月一共多少、月均多少（空月份也算进分母，好和「每月大概花多少」对上） */
export function trendSummary(series) {
  const rows = series || [];
  const totalCents = rows.reduce((sum, r) => sum + (Number(r && r.cents) || 0), 0);
  return { totalCents, avgCents: rows.length ? Math.round(totalCents / rows.length) : 0 };
}
```

```js
/** 左栏第三张：近 6 个月趋势（柱状）。页签切支出 / 收入，悬停给完整金额。 */
function trendCard(txs) {
  const months = lastMonths(currentMonth(), TREND_MONTHS);
  const series = monthTrend(txs, months, trendType);
  const sum = trendSummary(series);
  const isIncome = trendType === "income";
  const head = `
    <div class="card-head">
      <h2>近 ${TREND_MONTHS} 个月</h2>
      <span class="hint">${
        sum.totalCents
          ? `${isIncome ? "收入" : "支出"}共 ${fmtMoney(sum.totalCents)} · 月均 ${fmtMoney(sum.avgCents)}`
          : ""
      }</span>
    </div>
    <div class="fin-ring-tabs">
      <button class="fin-tab${isIncome ? "" : " active"}" data-act="trend-type" data-type="expense">支出</button>
      <button class="fin-tab${isIncome ? " active" : ""}" data-act="trend-type" data-type="income">收入</button>
    </div>`;
  if (!sum.totalCents) {
    return `${head}${emptyState(
      `近 ${TREND_MONTHS} 个月还没有账目`,
      "记上几笔，这里就能看出高低。",
      "",
      "money"
    )}`;
  }
  // 手写 SVG（不引图表库）：固定坐标系，柱宽按月份数平分
  const W = 600;
  const H = 180;
  const gap = 14;
  const bw = (W - gap * (series.length - 1)) / series.length;
  const baseY = H - 24;          // 柱子底线（下面留一行写月份）
  const chartH = baseY - 30;     // 柱子最高到哪儿（上面留一行写金额）
  const last = series[series.length - 1].month;
  const bars = series
    .map((row, i) => {
      const h = row.cents ? Math.max(3, Math.round(chartH * row.ratio)) : 0;
      const x = i * (bw + gap);
      const y = baseY - h;
      const mid = (x + bw / 2).toFixed(1);
      const tip = `${row.month} · ${isIncome ? "收入" : "支出"} ${fmtMoney(row.cents)}`;
      return `<g class="trend-bar${row.month === last ? " now" : ""}">
        <title>${esc(tip)}</title>
        <rect x="${x.toFixed(1)}" y="${y}" width="${bw.toFixed(1)}" height="${h}" rx="5"></rect>
        ${
          row.cents
            ? `<text class="trend-num" x="${mid}" y="${Math.max(13, y - 7)}" text-anchor="middle">${esc(
                fmtMoneyShort(row.cents)
              )}</text>`
            : ""
        }
        <text class="trend-mon" x="${mid}" y="${baseY + 17}" text-anchor="middle">${esc(
          Number(row.month.slice(5)) + "月"
        )}</text>
      </g>`;
    })
    .join("");
  return `${head}
    <svg class="fin-trend ${isIncome ? "is-income" : "is-expense"}" viewBox="0 0 ${W} ${H}"
      role="img" aria-label="${isIncome ? "收入" : "支出"}近 ${TREND_MONTHS} 个月走势">
      <line class="trend-base" x1="0" y1="${baseY}" x2="${W}" y2="${baseY}"></line>
      ${bars}
    </svg>
    ${
      last === currentMonth()
        ? `<p class="hint">${esc(monthCN(last))}是到 ${esc(todayStr().slice(5))} 为止的数，这个月还没过完。</p>`
        : ""
    }`;
}
```
