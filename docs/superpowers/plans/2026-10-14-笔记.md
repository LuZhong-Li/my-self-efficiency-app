# 笔记模块（富文本 / Markdown）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增第 11 个模块「笔记」：一篇一篇的 Markdown 笔记，能贴图、贴标签、全文搜到；左边列表、右边写，右上角「预览」看渲染结果。

**Tech Stack:** 原生 HTML/CSS/ES 模块（无框架、无构建、零外链）；Python 3 标准库小服务；测试用 Node 22+ 内置 `fetch` / `WebSocket` 直连 CDP 驱动无头 Edge。

**Spec:** `docs/superpowers/specs/2026-10-14-笔记-design.md`

- **一个任务一个提交**，commit message 用中文；直接提交到 `main`。

```js
/* 笔记的 Markdown 渲染器：纯函数，只吃字符串、吐 HTML 片段。
 *
 * 为什么要自己写一个：这个项目零外链、不装依赖，引不了现成的库。
 * 好在个人笔记用不上完整的 CommonMark —— 标题 / 粗斜 / 代码 / 列表 / 引用 /
 * 链接图片这几样就够（表格、脚注、公式都不做，见设计文档第 7 节）。
 *
 * 安全规矩（这一条最要紧，顺序不能反）：
 *   1. **先把整段文字转义**，再往里插我们自己生成的标签 ——
 *      所以笔记里写 <script> 只会原样显示成文字；
 *   2. 链接只认 http / https 这两种协议、页内锚点（#）和本地附件路径（attachments/…）；
 *      其它一律不生成 <a>，原样当文字显示（javascript: 那种就这么被挡住的）；
 *   3. 生成的标签都带自己的 class（md-h1 / md-p / md-code…），样式在 style.css 里。
 */

/** 转义 HTML 里那几个要紧的字符（和 store.js 的 esc 一个实现，纯模块 import 不了那个） */
function esc(text) {
  return String(text ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

/**
 * 地址白名单：https 开头的、页内锚点（#）、本地附件（attachments/）三种认，
 * 其它给空串（调用方按「原样显示」处理）。
 * 注意：传进来的是**已经转义过**的文本，所以这里不能再转义一次（否则 &amp; 会变 &amp;amp;）。
 */
function safeUrl(url) {
  const u = String(url || "").trim();
  if (/^https?:\/\//i.test(u)) return u;
  if (u.startsWith("#") || u.startsWith("attachments/")) return u;
  return "";
}

/**
 * 行内语法。**入参必须是已经转义过的文本**。
 * 顺序有讲究：行内代码先抠出来（里面的星号不当语法），图片在链接前面（只差一个感叹号）。
 */
function inline(text) {
  const codes = [];
  let out = text.replace(/`([^`]+)`/g, (m, code) => {
    codes.push(code);
    return "\u0000" + (codes.length - 1) + "\u0000";
  });
  out = out.replace(/!\[([^\]]*)\]\(([^)\s]+)\)/g, (m, alt, src) => {
    const url = safeUrl(src);
    if (!url) return m;                       // 地址认不出就原样留着
    return `<img class="md-img" src="${url}" alt="${alt}">`;
  });
  out = out.replace(/\[([^\]]+)\]\(([^)\s]+)\)/g, (m, label, href) => {
    const url = safeUrl(href);
    if (!url) return m;
    return `<a class="md-link" href="${url}" target="_blank" rel="noopener">${label}</a>`;
  });
  out = out.replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>");
  out = out.replace(/(^|[^*])\*([^*]+)\*/g, "$1<em>$2</em>");
  out = out.replace(/~~([^~]+)~~/g, "<del>$1</del>");
  // 最后把代码占位符换回来
  return out.replace(/\u0000(\d+)\u0000/g, (m, i) => `<code class="md-code">${codes[Number(i)]}</code>`);
}

/** 一段普通文字（可能含多个单回车）→ 一个 <p> */
function paraHtml(lines) {
  const body = inline(esc(lines.join("\n"))).replace(/\n/g, "<br>");
  return `<p class="md-p">${body}</p>`;
}

/** 引用块（里面也可以有多行） */
function quoteHtml(lines) {
  const body = inline(esc(lines.join("\n"))).replace(/\n/g, "<br>");
  return `<blockquote class="md-quote">${body}</blockquote>`;
}

/**
 * 整段 Markdown → HTML。逐行扫的「状态机」：
 * 攒着当前这一段 / 这份列表 / 这段引用 / 这个代码块，遇到换结构的地方就收口。
 */
export function renderMarkdown(text) {
  const src = String(text == null ? "" : text).replace(/\r\n?/g, "\n");
  if (!src.trim()) return "";
  const out = [];
  let para = [];
  let list = null;   // { tag: "ul" | "ol", items: [] }
  let quote = [];
  let code = null;   // 代码块里攒的行

  const flushPara = () => {
    if (para.length) {
      out.push(paraHtml(para));
      para = [];
    }
  };
  const flushList = () => {
    if (!list) return;
    const items = list.items.map((t) => `<li class="md-li">${inline(esc(t))}</li>`).join("");
    out.push(`<${list.tag} class="md-${list.tag}">${items}</${list.tag}>`);
    list = null;
  };
  const flushQuote = () => {
    if (quote.length) {
      out.push(quoteHtml(quote));
      quote = [];
    }
  };
  const flushAll = () => {
    flushPara();
    flushList();
    flushQuote();
  };

  for (const line of src.split("\n")) {
    // 代码块里：原样攒着，只找收尾的那三个反引号
    if (code) {
      if (/^```\s*$/.test(line)) {
        out.push(`<pre class="md-pre"><code>${esc(code.join("\n"))}</code></pre>`);
        code = null;
      } else {
        code.push(line);
      }
      continue;
    }
    if (/^```/.test(line)) {
      flushAll();
      code = [];
      continue;
    }
    if (!line.trim()) {
      flushAll();
      continue;
    }
    const head = /^(#{1,6})\s+(.*)$/.exec(line);
    if (head) {
      flushAll();
      const level = head[1].length;
      out.push(`<h${level} class="md-h${level}">${inline(esc(head[2].trim()))}</h${level}>`);
      continue;
    }
    if (/^(?:-{3,}|\*{3,})$/.test(line.trim())) {
      flushAll();
      out.push('<hr class="md-hr">');
      continue;
    }
    const quoted = /^>\s?(.*)$/.exec(line);
    if (quoted) {
      flushPara();
      flushList();
      quote.push(quoted[1]);
      continue;
    }
    const bullet = /^[-*+]\s+(.*)$/.exec(line);
    if (bullet) {
      flushPara();
      flushQuote();
      if (!list || list.tag !== "ul") {
        flushList();
        list = { tag: "ul", items: [] };
      }
      list.items.push(bullet[1]);
      continue;
    }
    const numbered = /^\d+[.)]\s+(.*)$/.exec(line);
    if (numbered) {
      flushPara();
      flushQuote();
      if (!list || list.tag !== "ol") {
        flushList();
        list = { tag: "ol", items: [] };
      }
      list.items.push(numbered[1]);
      continue;
    }
    // 剩下的都是普通正文：列表 / 引用到此为止
    flushList();
    flushQuote();
    para.push(line.trim());
  }
  // 收尾：没闭合的代码块也照原样收住
  if (code) out.push(`<pre class="md-pre"><code>${esc(code.join("\n"))}</code></pre>`);
  flushAll();
  return out.join("");
}
```

```js
/* 笔记：左边列表、右边写。正文就是 Markdown 源码，右上角「预览」看渲染结果。
 *
 * 为什么不开弹窗：弹窗是给「填几条字段」用的，写长文该像本地编辑器 ——
 * 打字时静默保存（不重画，光标不跳），离开输入框时正常保存并刷新左边列表。
 * 渲染是纯逻辑（markdown.js），图片走现成的附件组件（目录 note，早就预留好了）。
 */

import { touch, table, esc, uid, moveToTrash } from "./store.js";
import { pageHeader, bindFresh, emptyState } from "./ui.js";
import { askConfirm, toast } from "./dialog.js";
import { icon } from "./icons.js";
import { tagsOfRow, tagColor } from "./tag-calc.js";

let currentId = null;   // 正在写哪一条（只记在内存里：刷新回到「没选」）
let filterText = "";    // 左边列表的筛选词（也只记在内存里）

export function renderNote(root, sub = "") {
  const notes = table("notes");
  // 地址 #note/<id>：从搜索点过来直接打开那一条
  if (sub && notes.some((n) => n.id === sub)) currentId = sub;
  const cur = notes.find((n) => n.id === currentId) || null;
  root.innerHTML = `
    ${pageHeader("note", `<span class="date-chip">${notes.length} 篇</span>`)}
    <div class="note-layout">
      <section class="card note-side">
        <div class="card-head">
          <h2>笔记</h2>
          <div class="card-tools">
            <button class="btn primary small" data-act="note-new">${icon("plus", 14)}新笔记</button>
          </div>
        </div>
        <input id="note-filter" class="grow note-filter" maxlength="60" autocomplete="off"
          placeholder="筛选标题 / 正文 / 标签…" value="${esc(filterText)}">
        <div id="note-list-host">${listHtml(notes)}</div>
      </section>
      <section class="card note-main">${editorHtml(cur)}</section>
    </div>
  `;
  bindFresh(root, { click: onClick, input: onInput, change: onChange });
}

/** 正文第一行（没标题时用它顶上列表和搜索结果） */
function firstLine(body) {
  const line = String(body || "").split("\n").find((l) => l.trim()) || "";
  return line.replace(/^#+\s*/, "").replace(/^[-*+]\s+/, "").trim().slice(0, 40);
}

/** 列表上那句摘要：正文去掉 Markdown 记号，掐前 60 个字 */
function subText(n) {
  return String(n.body || "").replace(/[#>*`~]/g, " ").replace(/\s+/g, " ").trim().slice(0, 60);
}

/** 右边那块：选了笔记就是编辑器，没选就一句提示 */
function editorHtml(cur) {
  if (!cur) {
    return `<div class="note-pick-hint">${emptyState(
      "选一条，或者新建一条",
      "左边点一条笔记开始写；右上角「新笔记」开一条新的。",
      "",
      "pencil"
    )}</div>`;
  }
  return `
    <div class="note-editor-head">
      <input id="note-title" class="grow note-title" maxlength="80" autocomplete="off"
        placeholder="标题（可以空着）" value="${esc(cur.title || "")}">
      <button class="link danger" data-act="note-del">删除</button>
    </div>
    <textarea id="note-body" class="note-text"
      placeholder="写点什么…（支持 Markdown：标题 #、列表 -、加粗 **、代码块 三个反引号）">${esc(cur.body || "")}</textarea>
    <p class="note-hint">打字会自动存；离开输入框会刷新左边列表。</p>
  `;
}

/** 左边那一列：按更新时间倒序（越新越上面），再按筛选词过一遍 */
function listHtml(notes) {
  const kw = filterText.trim().toLowerCase();
  const rows = notes
    .slice()
    .sort((a, b) =>
      String(a.updatedAt || a.createdAt) < String(b.updatedAt || b.createdAt) ? 1 : -1)
    .filter((n) => {
      if (!kw) return true;
      const hay = [n.title, n.body, tagsOfRow(n).join(" ")].join(" ").toLowerCase();
      return hay.includes(kw);
    });
  if (!rows.length) {
    return notes.length
      ? `<p class="empty">没有符合「${esc(filterText.trim())}」的笔记。</p>`
      : `<p class="empty note-empty">还没有笔记。点右上角「新笔记」写第一条。</p>`;
  }
  return `<ul class="items note-list">${rows.map(noteRow).join("")}</ul>`;
}

/** 列表上的一行（整行可点 = 选这条） */
function noteRow(n) {
  const title = n.title || firstLine(n.body) || "（空笔记）";
  const chips = tagsOfRow(n)
    .slice(0, 3)
    .map((t) => `<span class="chip tag-chip" data-c="${tagColor(t)}">${esc(t)}</span>`)
    .join("");
  return `
    <li class="item note-row${n.id === currentId ? " active" : ""}"
        data-act="note-pick" data-id="${esc(n.id)}">
      <span class="i-title nr-title">${esc(title)}</span>
      <span class="i-note nr-sub">${esc(subText(n))}</span>
      ${chips}
    </li>`;
}

function findNote(id) {
  return table("notes").find((n) => n && n.id === id) || null;
}

/** 记一下「刚改过」（列表按这个倒序） */
function stamp(n) {
  n.updatedAt = new Date().toISOString();
  return n;
}

/** 新建一条空的并选中它 */
function newNote() {
  const now = new Date().toISOString();
  const row = {
    id: uid(), title: "", body: "", tags: [], imagePaths: [],
    createdAt: now, updatedAt: now,
  };
  table("notes").push(row);
  currentId = row.id;
  touch(true);
}

async function deleteNote(row) {
  const ok = await askConfirm({
    title: "删掉这篇笔记？",
    message: `${row.title || firstLine(row.body) || "（空笔记）"}\n\n` +
      "会放进回收站，误删可以去「数据与设置」找回。",
    confirmLabel: "删除",
    danger: true,
  });
  if (!ok) return;
  moveToTrash("notes", row, row.title || firstLine(row.body) || "空笔记");
  if (currentId === row.id) currentId = null;
  touch(true);
  toast("已移入回收站");
}

/** 打字时静默保存：只写数据、不重画（光标不会跳） */
function onInput(e) {
  // 筛选框：只重画左边那一列（整页重画会把正在打字的框冲掉）
  if (e.target.id === "note-filter") {
    filterText = e.target.value;
    const host = document.getElementById("note-list-host");
    if (host) host.innerHTML = listHtml(table("notes"));
    return;
  }
  const note = findNote(currentId);
  if (!note) return;
  if (e.target.id === "note-title") note.title = e.target.value;
  else if (e.target.id === "note-body") note.body = e.target.value;
  else return;
  stamp(note);
  touch(false, true);
}

/** 离开输入框：正常保存一次并重画（列表的标题 / 摘要 / 排序跟着变） */
function onChange(e) {
  const note = findNote(currentId);
  if (!note) return;
  if (e.target.id === "note-title") note.title = e.target.value;
  else if (e.target.id === "note-body") note.body = e.target.value;
  else return;
  stamp(note);
  touch(true);
}

async function onClick(e) {
  const btn = e.target.closest("[data-act]");
  if (!btn) return;
  const act = btn.dataset.act;
  if (act === "note-new") {
    newNote();
    return;
  }
  if (act === "note-pick") {
    currentId = btn.dataset.id;
    touch(true);
    return;
  }
  if (act === "note-del") {
    const note = findNote(currentId);
    if (note) await deleteNote(note);
  }
}

async function onChange(e) {
  const note = findNote(currentId);
  if (!note) return;
  if (e.target.id === "note-title") note.title = e.target.value;
  else if (e.target.id === "note-body") note.body = e.target.value;
  else if (e.target.id === "note-tags") {
    const err = tagInputError(e.target.value);
    if (err) {
      toast(err, "err");
      return;
    }
    note.tags = normalizeTags(e.target.value);
  } else return;
  stamp(note);
  touch(true);
  // 离开输入框时顺手把待存的图存掉（贴完图直接点别处也不会丢）
  if (attach && attach.files.length) await saveImages();
}

/** 把待存的那几张图真写进 附件目录/note/，再记到这条笔记上 */
async function saveImages() {
  const note = findNote(currentId);
  if (!note || !attach || !attach.files.length) return;
  let uploaded = [];
  try {
    uploaded = await uploadPending(attach);
  } catch (err) {
    toast("图片没存下：" + err.message, "err");
    return;
  }
  commitUploads(attach, uploaded);
  note.imagePaths = attach.paths.slice();
  stamp(note);
  touch(true);
  toast("图片已存下");
}
